<script type="text/javascript">
    const loadStaffChangeCount = async () => {
        try {
            let records = await $j.ajax({
                url: '/admin/staff_change/json/staffCountsData.json',
                method: 'GET',
                data: {
                    curSchoolID: '~(curschoolid)',
                    calendarYear: '~(f.currentdate;format= YYYY)'
                }
            })

            if (typeof psUtils !== 'undefined' && psUtils.htmlEntitiesToCharCode) {
                records = psUtils.htmlEntitiesToCharCode(records)
            }
            if (typeof records === 'string') {
                records = JSON.parse(records)
            }

            const countRecord = Array.isArray(records) ? records[0] : records
            return Number(countRecord && countRecord.total_remaining) || 0
        } catch (err) {
            console.log(err)
            return 0
        }
    }
</script>

<script type="text/javascript">
    const findStaffNavItemById = (itemId, startWith) => {
        if (typeof appNavItems == 'undefined') return
        let items = startWith ? startWith.children || [] : appNavItems
        let matchedItem
        items.forEach(item => {
            if (item.id == itemId) matchedItem = item
        })
        if (matchedItem) return matchedItem
        //if not found, search children
        else {
            items.forEach(item => {
                if (item.children && !matchedItem) matchedItem = findStaffNavItemById(itemId, item)
            })
            if (matchedItem) return matchedItem
        }
    }
    //update count on link and add icon
    const updateStaffCount = (linkId, count) => {
        let navItem = findStaffNavItemById(linkId)
        if (!navItem || count == 0) {
            $j('#staffCountSpan').remove()
            $j('#staffCountLi').removeClass('countMargin')
        } else {
            navItem.name = `${navItem.name} (${count})`
            $j('#staffCountSpan').text(count)
            $j('#tools2').prepend($j('#staffCount').html())
        }
        $j('#staffCount').remove()
    }

    $j(() =>
        (async () => {
            const staffCount = await loadStaffChangeCount()
            updateStaffCount('staff-changes-list', staffCount)
        })()
    )
</script>
